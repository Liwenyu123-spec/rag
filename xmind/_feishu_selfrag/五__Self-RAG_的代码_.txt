"""
Self-RAG（自我反思式 RAG）- LlamaIndex 实现

核心思想：
    引入“批评家”角色（由 qwen-plus/qwen-max 扮演），在传统的 RAG 检索/生成流程中插入四个关键的反思标记（Reflection Tokens）：
    1. Retrieve（要不要检索？）：判断问题是否需要外部知识。
    2. ISREL（资料相关吗？）：判断检索到的片段是否与问题真正相关。
    3. ISSUP（有证据支持吗？）：判断生成的回答是否忠实于检索到的资料（抑制幻觉）。
    4. ISUSE（回答有用吗？）：评估最终回答对用户的有用程度。

技术实现：
    通过继承 LlamaIndex 的 `CustomQueryEngine`，将上述自我反思逻辑封装成一个标准的查询引擎，
    使其可以像普通 QueryEngine 一样被调用，但内部具备自我校验和修正能力。
"""

import os

from llama_index.core import (
    Settings,
    SimpleDirectoryReader,
    VectorStoreIndex,
)
from llama_index.core.base.base_retriever import BaseRetriever
from llama_index.core.llms import LLM
from llama_index.core.query_engine import CustomQueryEngine
from llama_index.embeddings.dashscope import DashScopeEmbedding
from llama_index.llms.dashscope import DashScope
from llama_index.core.base.embeddings.base import BaseEmbedding
from typing import List, Any
from pydantic import PrivateAttr

# ===================== 1. 全局配置（通义千问） =====================
# 从环境变量获取 API Key，确保安全性
api_key = os.getenv("DASHSCOPE_API_KEY")

class DashScopeEmbeddingWithBatch(BaseEmbedding):

# 配置全局嵌入模型
Settings.embed_model = DashScopeEmbeddingWithBatch(batch_size=10)

# 配置全局 LLM（使用通义千问最新模型，低温以减少随机性，适合逻辑判断）
Settings.llm = DashScope(
    model_name="qwen3.7-max",  # 使用高性能版本作为“批评家”和生成器
    api_key=api_key,
    temperature=0.1,           # 低温度确保输出稳定，特别是对于分类任务（YES/NO）
