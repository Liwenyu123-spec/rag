        ).upper()

        )

    # ---- 自我修正：当支持度不足时，强制贴合资料重写 ----
    def _correct(self, query: str, context: str, answer: str) -> str:
        """
        基于原始资料对回答进行修正，消除幻觉，确保忠实于原文。
        
        Returns:
            修正后的回答
        """
        return self._judge(
            "你是严谨的 Self-RAG 修正器。请基于资料重写回答，确保：\n"
            "1) 完全依据资料，不臆造；2) 无幻觉；3) 完整回答问题。\n\n"
            f"资料：{context}\n问题：{query}\n原回答：{answer}\n\n修正后的回答："
        )

    # ===================== Self-RAG 主流程 =====================
    def custom_query(self, query_str: str) -> str:
        """
        执行 Self-RAG 的核心查询逻辑。
        
        流程：
        1. [Retrieve] 判断是否需要检索。
        2. 若需要，执行检索。
        3. [ISREL] 过滤不相关的检索结果。
        4. 生成初始回答。
        5. [ISSUP] 检查支持度，若不足则调用 [_correct] 修正。
        6. [ISUSE] 评估最终质量（仅用于日志/监控）。
        
        Args:
            query_str: 用户查询字符串
            
        Returns:
            最终生成的回答
        """
        if self.verbose:
            print(f"\n===== Self-RAG 处理：{query_str} =====")

        # [Step 1: Retrieve] 判断是否需要检索
        if not self._need_retrieve(query_str):
            if self.verbose:
                print("[Retrieve=NO] 无需检索，直接回答")
            # 如果不需要检索，直接让 LLM 依靠内部知识回答
            return self.llm.complete(query_str).text.strip()

        # 执行检索（使用 LlamaIndex 配置的检索器）
        nodes = self.retriever.retrieve(query_str)
        if not nodes:
            return "未检索到相关信息。"

六、对self-RAG的进一步理解
Self-RAG 更彻底的实现是在 token 级别让 LLM 一边生成、一边输出反思标记来决定是否检索（依赖专门微调的 selfrag_llama2_7b 模型）。上一节的代码用通用 LLM 在"判断点"显式询问来等价复现这套闭环，下面进一步拆解其训练与推理原理。
1、反思令牌（Reflection Tokens）：模型的"内心独白"
SELF-RAG的核心创新是引入了四种特殊的"反思令牌"，让模型在生成文本的同时，输出自己的思考过程。这四种令牌就像模型的"内心独白"：
•
Retrieve（检索决策）：是否需要查资料？
◦
通俗地说：就像我们在写作文时判断"这句话需要查资料确认一下吗？"
◦
场景：回答"苹果是什么"不需要查资料，但"iPhone 15的发布日期是哪天"就需要
•
ISREL（相关性评估）：检索的资料有用吗？
◦
通俗地说：就像找资料时判断"这篇文章和我关心的问题相关吗？"
◦
场景：问"Python编程"时，检索到"C++教程"就是不相关的
•
ISSUP（支持度评估）：生成的内容有证据支持吗？
◦
通俗地说：就像法官判决时问"有确凿证据支持这个说法吗？"
◦
场景：说"月球距离地球38万公里"需要有天文数据支撑，但"月亮很美"是主观感受
•
ISUSE（有用性评估）：这个回答对用户有帮助吗？
◦
通俗地说：就像老师批改作业时想"这个回答解决了学生的问题吗？"
◦
场景：用户问"怎么做蛋糕"，回答"我饿了"虽然真实但不相关
2、两阶段训练：从"外部批评"到"自我监督"
SELF-RAG的训练分两个阶段，巧妙地利用了qwen3.7-max作为"老师"，最终实现模型的"自学成才"：
第一阶段：训练批评家模型（Critic Model）
•
用qwen3.7-max生成反思令牌作为监督信号（例如让qwen3.7-max判断"这个问题需要查资料吗"）
•
训练一个较小的模型（如Llama2-7B）模仿qwen3.7-max的判断逻辑
•
通俗地说：就像让qwen3.7-max批改作业，然后训练一个助教学会qwen3.7-max的批改标准
第二阶段：训练生成模型（Generator Model）
•
在原始训练数据中，根据批评家的判断插入反思令牌和检索的文档
•
训练模型同时预测"应该说什么"和"应该如何评价自己"
•
关键点：检索的文档不参与损失计算（masked），只作为上下文信息
•
通俗地说：就像给学生看例题时，同时展示"正确答案"和"解题思路"，让他们学会边写边反思
3、推理时的自适应机制：动态调整行为
在推理阶段，SELF-RAG展现出惊人的灵活性：
自适应检索（Adaptive Retrieval）
•
模型根据Retrieve令牌的概率动态决定是否检索
•
可以设置阈值：概率超过多少时才触发检索
•
通俗地说：就像根据题目难度决定是否翻参考书，简单题直接答，难题才查资料
