    def custom_query(self, query_str: str) -> str:
        if self.verbose:
            print(f"\n===== Corrective RAG 处理：{query_str} =====")

        # 步骤1：检索
        nodes = self.retriever.retrieve(query_str)
        if self.verbose:
            print(f"[1] 初始检索到 {len(nodes)} 篇文档")

        # 步骤2：相关性评估 + 过滤（CRAG 核心）
        if self.verbose:
            print("[2] 文档相关性过滤中...")
        relevant = self._filter_relevant(query_str, nodes)

        # 步骤3：全部不相关 → 重写查询，重检索（内部修正）
        if not relevant:
            if self.verbose:
                print("[3] 信息不足，重写查询并重检索...")
            new_query = self.llm.complete(
                REWRITE_PROMPT.format(query=query_str)
            ).text.strip()
            if self.verbose:
                print(f"   改写后问题：{new_query}")
            relevant = self._filter_relevant(query_str, self.retriever.retrieve(new_query))

        if not relevant:
            return "知识库中没有足够信息回答该问题。"

        # 步骤4：生成
        if self.verbose:
            print(f"[4] 用 {len(relevant)} 篇相关文档生成回答...")
        context = "\n---\n".join(relevant)
        return self.llm.complete(
            ANSWER_PROMPT.format(query=query_str, context=context)
        ).text

四、核心知识点总结
1.
Corrective RAG = 普通 RAG + 检索质量评估 + 自动修正
2.
评估器把检索结果分为 Correct / Ambiguous / Incorrect 三档（工程实现常简化为逐篇 yes/no）
3.
核心修正手段 = 知识精炼（内部去噪）+ Web 搜索（外部补充）
4.
能大幅缓解幻觉、过时信息、无关检索问题，对"私有知识库 + 公共知识"混合场景效果尤其好
五、示例代码（完整版：LlamaIndex 官方 CRAG Workflow + Web 搜索）
基础版只能在"内部知识库"里打转——文档不相关时只是重写查询再查一遍同一个库。
完整 CRAG 的关键在于：当内部文档不足时，转向外部 Web 搜索补充知识。
LlamaIndex 官方提供了开箱即用的 CorrectiveRAGWorkflow（基于事件驱动的 Workflow），完整实现了 CRAG 流程：检索 → 逐篇相关性评估 → 只要有文档被判"不相关"就重写查询并调用 Tavily Web 搜索 → 合并"内部相关文本 + 外部搜索结果"生成答案。

Web 搜索这里用 Tavily（专为 LLM 设计的搜索 API），需要在 tavily.com 申请一个免费 API Key。
