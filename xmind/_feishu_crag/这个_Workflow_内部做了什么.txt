# ===================== 4. Corrective RAG Workflow =====================
class CorrectiveRAGWorkflow(Workflow):
    @step
    async def ingest(self, ctx: Context, ev: StartEvent) -> StopEvent | None:
        """传入 documents 时：建索引并返回 index。"""
        documents = ev.get("documents")
        if documents is None:
            return None
        index = VectorStoreIndex.from_documents(documents)
        return StopEvent(result=index)

    @step
    async def prepare_for_retrieval(self, ctx: Context, ev: StartEvent) -> PrepEvent | None:
        """传入 query_str/index 时：把 LLM、index、Tavily 工具存进上下文。"""
        query_str = ev.get("query_str")
        if query_str is None:
            return None
        await ctx.store.set("index", ev.get("index"))
        await ctx.store.set("tavily_tool", TavilyToolSpec(api_key=ev.get("tavily_ai_apikey")))
        await ctx.store.set("query_str", query_str)
        return PrepEvent()

    @step
    async def retrieve(self, ctx: Context, ev: PrepEvent) -> RetrieveEvent:
        """检索候选节点。"""
        query_str = await ctx.store.get("query_str")
        index = await ctx.store.get("index")
        nodes = index.as_retriever(similarity_top_k=5).retrieve(query_str)
        await ctx.store.set("retrieved_nodes", nodes)
        return RetrieveEvent(retrieved_nodes=nodes)

    @step
    async def eval_relevance(self, ctx: Context, ev: RetrieveEvent) -> RelevanceEvalEvent:
        """检索评估器：逐篇让 LLM 打 yes/no（CRAG 核心）。"""
        query_str = await ctx.store.get("query_str")
        results = []
        for node in ev.retrieved_nodes:
            resp = await Settings.llm.acomplete(
                RELEVANCY_PROMPT.format(context_str=node.get_content(), query_str=query_str)
            )
            results.append(resp.text.lower().strip())
        await ctx.store.set("relevancy_results", results)
        return RelevanceEvalEvent(relevant_results=results)

    @step
    async def extract_relevant_texts(self, ctx: Context, ev: RelevanceEvalEvent) -> TextExtractEvent:
        """只保留被判 yes 的文档文本。"""
        nodes = await ctx.store.get("retrieved_nodes")
        relevant = [
            nodes[i].get_content()
            for i, r in enumerate(ev.relevant_results)
            if r == "yes"
        ]
        return TextExtractEvent(relevant_text="\n".join(relevant))

    @step
    async def transform_query(self, ctx: Context, ev: TextExtractEvent) -> QueryEvent:
        """只要出现一个 no：重写查询 + Tavily Web 搜索补充外部知识。"""
        results = await ctx.store.get("relevancy_results")
        query_str = await ctx.store.get("query_str")
        search_text = ""
        if "no" in results:
            resp = await Settings.llm.acomplete(
                TRANSFORM_QUERY_PROMPT.format(query_str=query_str)
            )
            tavily_tool = await ctx.store.get("tavily_tool")
            search_results = tavily_tool.search(resp.text.strip(), max_results=5)
            search_text = "\n".join(r.text for r in search_results)
        return QueryEvent(relevant_text=ev.relevant_text, search_text=search_text)

    @step
    async def query_result(self, ctx: Context, ev: QueryEvent) -> StopEvent:
        """合并"内部相关文本 + 外部搜索文本"生成最终答案。"""
        query_str = await ctx.store.get("query_str")
        documents = [Document(text=ev.relevant_text + "\n" + ev.search_text)]
        query_engine = SummaryIndex.from_documents(documents).as_query_engine()
        return StopEvent(result=query_engine.query(query_str))
