"""
RAG 生成质量评估 - LlamaIndex 内置评估器
FaithfulnessEvaluator（忠实度，无需标准答案）
RelevancyEvaluator（相关性，无需标准答案）
CorrectnessEvaluator（正确性，需要标准答案 reference）
裁判模型用通义千问 qwen-plus。
"""

import os

from llama_index.core import Settings, SimpleDirectoryReader, VectorStoreIndex
from llama_index.core.evaluation import (
    CorrectnessEvaluator,
    FaithfulnessEvaluator,
    RelevancyEvaluator,
)
from llama_index.embeddings.dashscope import DashScopeEmbedding
from llama_index.llms.dashscope import DashScope

# ===================== 1. 全局配置（通义千问） =====================
api_key = os.getenv("DASHSCOPE_API_KEY")
# 评委模型温度设 0，保证打分稳定可复现
Settings.llm = DashScope(model_name="qwen3.7-max", temperature=0, api_key=api_key)
Settings.embed_model = DashScopeEmbedding(model_name="text-embedding-v4", api_key=api_key)

# ===================== 2. 构建一个待评估的 RAG =====================
documents = SimpleDirectoryReader(input_files=["./data/company_info.txt"]).load_data()
index = VectorStoreIndex.from_documents(documents)
query_engine = index.as_query_engine(similarity_top_k=3)

# ===================== 3. 三个内置评估器（都用 Settings.llm 当裁判） =====================
faithfulness = FaithfulnessEvaluator()   # 忠实度：答案是否被召回上下文支撑
relevancy = RelevancyEvaluator()         # 相关性：答案+上下文是否切题
correctness = CorrectnessEvaluator()     # 正确性：答案 vs 标准答案，打 1~5 分

# ===================== 4. 评估若干问题 =====================
qa_pairs = [
    ("公司标准上下班时间是什么？", "9:00-18:00，午休 12:00-13:00"),
    ("公司员工总数是多少？", "2000人"),
    ("公司主营业务是什么？", "AI软件开发、大数据服务、云计算平台"),
]

if __name__ == "__main__":
    print("=" * 70)
    print("RAG 生成质量评估（LlamaIndex 内置评估器 + 通义千问裁判）")
    print("=" * 70)

    for question, reference in qa_pairs:
        # 先让 RAG 生成回答（response 里同时带着召回的 source_nodes）
        response = query_engine.query(question)

        # 忠实度 / 相关性：直接传 response，评估器会自动取出 source_nodes 当上下文
        faith_result = faithfulness.evaluate_response(query=question, response=response)
        rel_result = relevancy.evaluate_response(query=question, response=response)\n01-RAG评估\n三、示例代码（LlamaIndex 内置评估器）：\n三、示例代码（LlamaIndex 内置评估器）：\n1、准备一份有干扰信息的文档\n1、准备一份有干扰信息的文档\ndata\company_info.txt\ndata\company_info.txt\n代码块
